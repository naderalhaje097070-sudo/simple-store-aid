def trace(events):

    latest = None
    for event in events:
        if latest is not None and event["at"] < latest:
            continue
        old = row or {}
        kind = event["type"]
        row = {
            "status": event["status"],
            "product_id": old.get("product_id") if old and kind != "created" else event["product_id"],
            "price_id": old.get("price_id") if old and kind != "created" else event["price_id"],
            "period_end": old.get("period_end") if old and kind == "canceled" else event.get("period_end") or old.get("period_end"),
            "cancel_at_period_end": old["cancel_at_period_end"] if old and kind != "updated" else event.get("cancel_at_period_end", False),
        }
        latest = event["at"]
    return row
