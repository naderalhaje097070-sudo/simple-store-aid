name: B2 scanner authorization
on:
  push:
    branches: ['integration/g0-g1-reconcile-2026-10-18']
    paths:
      - '.github/workflows/b2-scanner-auth.yml'
      - 'tests/account-lifecycle/*scanner*'
      - 'changes/b2-ac01/SCANNER_AUTH.sql'
      - 'supabase/migrations/*_scanner_auth_01.sql'
permissions:
  contents: read
jobs:
  prepare:
    runs-on: ubuntu-latest
    outputs:
      ready: ${{ steps.detect.outputs.ready }}
    steps:
      - uses: actions/checkout@v4
        with:
          persist-credentials: false
      - uses: supabase/setup-cli@v2
        with:
          version: '2.101.0'
      - id: detect
        shell: bash
        run: |
          set -euo pipefail
          test "$(supabase --version)" = '2.101.0'
          supabase migration new --help > "$RUNNER_TEMP/migration-new-help.txt"
          supabase migration up --help > "$RUNNER_TEMP/migration-up-help.txt"
          grep -q -- '--local' "$RUNNER_TEMP/migration-up-help.txt"
          count="$(find supabase/migrations -name '*_scanner_auth_01.sql' | wc -l)"
          if [[ "$count" == 0 ]]; then
            echo 'ready=false' >> "$GITHUB_OUTPUT"
            work="$RUNNER_TEMP/ac01-generator"
            mkdir -p "$work"
            supabase --workdir "$work" init > "$RUNNER_TEMP/generator-init.log" 2>&1
            supabase --workdir "$work" migration new scanner_auth_01 > "$RUNNER_TEMP/generator-new.log" 2>&1
            generated="$(find "$work/supabase/migrations" -name '*_scanner_auth_01.sql')"
            test -n "$generated"
            cp changes/b2-ac01/SCANNER_AUTH.sql "$generated"
            mkdir -p "$RUNNER_TEMP/generated-migration"
            cp "$generated" "$RUNNER_TEMP/generated-migration/"
            GENERATED_FILE="$generated" python - <<'PY'
          import hashlib,json,os
          from pathlib import Path
          p=Path(os.environ['GENERATED_FILE'])
          r={'filename':p.name,'sha256':hashlib.sha256(p.read_bytes()).hexdigest(),'cli':'2.101.0',
             'command':'supabase migration new scanner_auth_01','source':os.environ['GITHUB_SHA'],'database_started':False}
          (Path(os.environ['RUNNER_TEMP'])/'generated-migration/generator.json').write_text(json.dumps(r,indent=2)+'\n')
          print('AC01_GENERATOR '+json.dumps(r))
          PY
          elif [[ "$count" == 1 ]]; then
            cmp changes/b2-ac01/SCANNER_AUTH.sql supabase/migrations/*_scanner_auth_01.sql
            echo 'ready=true' >> "$GITHUB_OUTPUT"
          else
            exit 2
          fi
      - uses: actions/upload-artifact@v4
        if: steps.detect.outputs.ready == 'false'
        with:
          name: b2-cli-generated-migration
          path: ${{ runner.temp }}/generated-migration/
          retention-days: 7
  authorization:
    needs: prepare
    if: needs.prepare.outputs.ready == 'true'
    runs-on: ubuntu-latest
    timeout-minutes: 25
    strategy:
      fail-fast: false
      matrix:
        mode: [fresh, upgrade]
    env:
      EXPECTED_SOURCE: ${{ github.sha }}
      CONTRACT_MODE: ${{ matrix.mode }}
    steps:
      - uses: actions/checkout@v4
        with:
          persist-credentials: false
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
      - name: Oracle negatives
        run: python -m unittest discover -s tests/account-lifecycle -p test_scanner_auth_probe.py -v
      - uses: supabase/setup-cli@v2
        with:
          version: '2.101.0'
      - name: Discover pinned CLI operations
        run: |
          set -euo pipefail
          test "$(supabase --version)" = '2.101.0'
          supabase start --help > "$RUNNER_TEMP/start-help.txt"
          supabase stop --help > "$RUNNER_TEMP/stop-help.txt"
          supabase migration up --help > "$RUNNER_TEMP/up-help.txt"
          supabase db advisors --help > "$RUNNER_TEMP/advisors-help.txt"
          supabase migration list --help > "$RUNNER_TEMP/list-help.txt"
          grep -q -- '--local' "$RUNNER_TEMP/up-help.txt"
          grep -q -- '--no-backup' "$RUNNER_TEMP/stop-help.txt"
          sudo apt-get update -qq
          sudo apt-get install -y postgresql-client iptables > "$RUNNER_TEMP/prerequisites.log" 2>&1
      - name: Restrict probe UID before credentials
        shell: bash
        run: |
          set -euo pipefail
          sudo useradd --system --no-create-home --shell /usr/sbin/nologin ac01
          uid="$(id -u ac01)"
          sudo iptables -N AC01
          sudo iptables -A AC01 -o lo -d 127.0.0.1/32 -p tcp -m multiport --dports 54321,54322 -j ACCEPT
          sudo iptables -A AC01 -j REJECT --reject-with icmp-port-unreachable
          sudo iptables -I OUTPUT 1 -m owner --uid-owner "$uid" -j AC01
          sudo ip6tables -I OUTPUT 1 -m owner --uid-owner "$uid" -j REJECT
          sudo -u ac01 "$(command -v python)" - <<'PY'
          import socket
          try:
              with socket.create_connection(('1.1.1.1',443),timeout=3): pass
          except OSError:
              print('AC01_CREDENTIAL_FREE_EGRESS_REJECTED')
          else:
              raise SystemExit('AC01_FIREWALL_FAILED')
          PY
          sudo install -d -m 711 -o ac01 -g ac01 /tmp/monjez-ac01
          sudo cp -a "$GITHUB_WORKSPACE/." /tmp/monjez-ac01/repo
          sudo install -d -m 700 -o ac01 -g ac01 /tmp/monjez-ac01/evidence
          sudo chown -R ac01:ac01 /tmp/monjez-ac01
          sudo iptables -nvxL AC01 > "$RUNNER_TEMP/firewall-before.txt"
      - name: Start disposable full chain
        shell: bash
        run: |
          set -euo pipefail
          umask 077
          stack="$RUNNER_TEMP/monjez-ac01-stack"
          mkdir -p "$stack"
          supabase --workdir "$stack" init > "$RUNNER_TEMP/init-private.log" 2>&1
          mkdir -p "$stack/supabase/migrations"
          cp supabase/migrations/*.sql "$stack/supabase/migrations/"
          if [[ "$CONTRACT_MODE" == upgrade ]]; then
            rm "$stack"/supabase/migrations/*_scanner_auth_01.sql
          fi
          if ! supabase --workdir "$stack" start > "$RUNNER_TEMP/start-private.log" 2>&1; then
            echo 'AC01_STACK_START_FAILURE_PRIVATE_LOG_WITHHELD'
            exit 1
          fi
          supabase --workdir "$stack" status -o json > "$RUNNER_TEMP/status-private.json"
          sudo install -m 600 -o ac01 -g ac01 "$RUNNER_TEMP/status-private.json" /tmp/monjez-ac01/status.json
      - name: Baseline then populated local upgrade
        if: matrix.mode == 'upgrade'
        shell: bash
        run: |
          set -euo pipefail
          cd /tmp/monjez-ac01/repo
          sudo -u ac01 env -i PATH="$PATH" PYTHONDONTWRITEBYTECODE=1 \
            EXPECTED_SOURCE="$EXPECTED_SOURCE" MONGEZZ_DEPLOY_TARGET=ci MONGEZZ_E2E_WRITE_TEST=1 CI_ISOLATED_SUPABASE=1 \
            VITE_SUPABASE_URL=http://127.0.0.1:54321 VITE_SUPABASE_PUBLISHABLE_KEY=ac01-preflight-placeholder \
            PGHOST=127.0.0.1 PGPORT=54322 PGUSER=postgres PGDATABASE=postgres PGPASSWORD=postgres \
            python tests/account-lifecycle/scanner_auth_probe.py --expected-source "$EXPECTED_SOURCE" \
              --status-file /tmp/monjez-ac01/status.json --evidence-dir /tmp/monjez-ac01/evidence \
              --private-state /tmp/monjez-ac01/private-state.json --phase before
          cp supabase/migrations/*_scanner_auth_01.sql "$RUNNER_TEMP/monjez-ac01-stack/supabase/migrations/"
          supabase --workdir "$RUNNER_TEMP/monjez-ac01-stack" migration up --local > "$RUNNER_TEMP/upgrade-private.log" 2>&1
      - name: Actual authorization and invariance
        shell: bash
        run: |
          set -euo pipefail
          cd /tmp/monjez-ac01/repo
          sudo -u ac01 env -i PATH="$PATH" PYTHONDONTWRITEBYTECODE=1 \
            EXPECTED_SOURCE="$EXPECTED_SOURCE" MONGEZZ_DEPLOY_TARGET=ci MONGEZZ_E2E_WRITE_TEST=1 CI_ISOLATED_SUPABASE=1 \
            VITE_SUPABASE_URL=http://127.0.0.1:54321 VITE_SUPABASE_PUBLISHABLE_KEY=ac01-preflight-placeholder \
            PGHOST=127.0.0.1 PGPORT=54322 PGUSER=postgres PGDATABASE=postgres PGPASSWORD=postgres \
            python tests/account-lifecycle/scanner_auth_probe.py --expected-source "$EXPECTED_SOURCE" \
              --status-file /tmp/monjez-ac01/status.json --evidence-dir /tmp/monjez-ac01/evidence \
              --private-state /tmp/monjez-ac01/private-state.json --phase after
      - name: Local migration history and advisor availability
        shell: bash
        run: |
          set -euo pipefail
          supabase --workdir "$RUNNER_TEMP/monjez-ac01-stack" migration list --local > "$RUNNER_TEMP/migrations.txt"
          if grep -q -- '--local' "$RUNNER_TEMP/advisors-help.txt"; then
            supabase --workdir "$RUNNER_TEMP/monjez-ac01-stack" db advisors --local > "$RUNNER_TEMP/advisors-private.txt" 2>&1
          else
            echo 'AC01_OFFICIAL_ADVISOR_LOCAL_MODE_UNAVAILABLE_NO_REMOTE_FALLBACK'
          fi
      - name: Cleanup and safe final receipts
        if: always()
        shell: bash
        run: |
          set -euo pipefail
          out="$RUNNER_TEMP/ac01-evidence"
          mkdir -p "$out"
          if [[ -d /tmp/monjez-ac01/evidence ]]; then
            sudo cp -a /tmp/monjez-ac01/evidence/. "$out/"
            sudo chown -R "$(id -un):$(id -gn)" "$out"
          fi
          sudo rm -f /tmp/monjez-ac01/status.json /tmp/monjez-ac01/private-state.json
          rm -f "$RUNNER_TEMP/status-private.json" "$RUNNER_TEMP/start-private.log"
          if [[ -d "$RUNNER_TEMP/monjez-ac01-stack/supabase" ]]; then
            supabase --workdir "$RUNNER_TEMP/monjez-ac01-stack" stop --no-backup > "$RUNNER_TEMP/stop-private.log" 2>&1
          fi
          python - <<'PY'
          import json,os,re,subprocess
          from pathlib import Path
          root=Path(os.environ['RUNNER_TEMP']);out=root/'ac01-evidence'
          def rejects(path):
              return sum(int(x.split()[0]) for x in path.splitlines() if 'REJECT' in x)
          before=rejects((root/'firewall-before.txt').read_text())
          after=rejects(subprocess.check_output(['sudo','iptables','-nvxL','AC01'],text=True))
          uid=subprocess.check_output(['id','-u','ac01'],text=True).strip()
          v6=sum(int(x.split()[0]) for x in subprocess.check_output(['sudo','ip6tables','-nvxL','OUTPUT'],text=True).splitlines()
                 if 'REJECT' in x and ('owner UID match '+uid) in x)
          cleanup={'source':os.environ['EXPECTED_SOURCE'],'credential_free_canary_rejected':before>0,
                   'unexpected_ipv4_rejects':after-before,'ipv6_rejects':v6,
                   'status_removed':not Path('/tmp/monjez-ac01/status.json').exists(),
                   'state_removed':not Path('/tmp/monjez-ac01/private-state.json').exists(),
                   'stack_removed':'monjez-ac01-stack' not in subprocess.check_output(['docker','ps','--format','{{.Names}}'],text=True)}
          (out/'cleanup.json').write_text(json.dumps(cleanup,indent=2)+'\n')
          for p in out.rglob('*'):
              if p.is_file():
                  text=p.read_text()
                  assert not re.search(r'eyJ[A-Za-z0-9_-]{8,}\.[A-Za-z0-9_-]{8,}\.[A-Za-z0-9_-]{8,}|sb_(?:secret|publishable)_[A-Za-z0-9_-]{8,}',text),'Credential-like evidence rejected'
          print('AC01_CLEANUP '+json.dumps(cleanup))
          assert before>0 and after==before and v6==0 and cleanup['status_removed'] and cleanup['state_removed'] and cleanup['stack_removed']
          PY
      - uses: actions/upload-artifact@v4
        if: always()
        with:
          name: b2-scanner-${{ matrix.mode }}-attempt${{ github.run_attempt }}
          path: ${{ runner.temp }}/ac01-evidence/
          retention-days: 7
